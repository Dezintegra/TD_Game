## Парная разность побед Стратега

В каждой паре xA/xB равны 1 для S, иначе 0; d=xB−xA. При переходе A→B (Стратег со стороны 1 на 0):

- g=14 приобретённых побед: 13 B→S + 1 D→S;
- l=23 утраченные победы: 19 S→B + 4 S→D;
- g+l=37 расходящихся пар победа/непобеда; 23 пары — непобеда в обоих прогонах, 0 — победа в обоих;
- pA=23/60=38,33 %, pB=14/60=23,33 %;
- Δ=(g−l)/N=−9/60=−0,15, то есть −15 п.п.

Стандартная ошибка среднего парных разностей при N>1:

```text
SE = sqrt(((g+l) - N*Δ²) / (N*(N-1)))
   = sqrt((37 - 60*0.15²) / (60*59))
   = 0.1003524861 = 10.03524861 п.п.
Δ/SE = -1.49473128
```

Ошибка независимых долей sqrt(pA×(1−pA)/60+pB×(1−pB)/60) составляет около 8,32 п.п.; парная ошибка здесь больше. Профиль ни разу не выиграл в обоих прогонах на одном мире: ковариация бинарных побед отрицательна. Обещать выигрыш точности от парности до чтения переходов было бы неверно.

Точная двусторонняя биномиальная проверка на 37 расходящихся парах при равновероятных направлениях даёт 2×P[Binomial(37, 1/2)≤14]=0,187741559. Это проверка победа/непобеда, а не всей матрицы 3×3. Она не устраняет смену seed ИИ и не доказывает причинность. Стандартная ошибка и p описывают статистическую модель выборки миров; для этих фиксированных 60 детерминированных миров сами количества точны. Обобщение на другие миры и текущую версию игры этим анализом не установлено.

При N=0 доли и Δ не определены; при N<2 не определена SE. Нулевая SE не превращается в бесконечное число сигм; при отсутствии расходящихся пар и непустой выборке точное p=1. Эти случаи проверены отдельно, но в реальных данных не возникли.

## Воспроизводимый расчёт и проверки

Сохранить следующий блок как .matchlog/compare-seed-paired-side-swaps.mjs в назначенном дереве. Скрипт читает базы только для чтения и пишет два промежуточных файла внутри .matchlog. Он предназначен именно для этих исторических источников: контрольные суммы исходов и равенство полных и парных маргинальных сумм — утверждения об этой паре прогонов. На других данных проверки могут остановить выполнение; это не универсальный инструмент анализа.

import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { createReadStream, readFileSync, writeFileSync } from 'node:fs';
import { DatabaseSync } from 'node:sqlite';

const baseline = 'baseline-2026-08';
const strategist = 'strategist-2026-08';
const labels = ['B', 'D', 'S'];
const sql = `select match_id, kind, world_seed, ai_seed_0, ai_seed_1,
  profile_0, profile_1, git_sha, git_dirty, ticks, winner, end_reason
  from match order by world_seed, match_id`;
const hash = async (path) => {
  const h = createHash('sha256');
  for await (const chunk of createReadStream(path)) h.update(chunk);
  return h.digest('hex');
};
const outcome = (m) => {
  if (m.kind !== 'arena' || !Number.isInteger(m.ticks) || m.ticks <= 0)
    throw new Error('not a completed arena match');
  if (m.end_reason === 'timeout' && m.winner === null) return 'D';
  if (m.end_reason !== 'base-destroyed' || ![0, 1].includes(m.winner))
    throw new Error('invalid winner/end_reason');
  const profile = m[`profile_${m.winner}`];
  if (profile === strategist) return 'S';
  if (profile === baseline) return 'B';
  throw new Error('unknown winner profile');
};
const group = (rows) => {
  const map = new Map();
  for (const row of rows) {
    const list = map.get(row.world_seed) ?? [];
    list.push(row);
    map.set(row.world_seed, list);
  }
  return map;
};
const pair = (a, b) => {
  const maps = [group(a), group(b)];
  const seeds = [...new Set([...maps[0].keys(), ...maps[1].keys()])].sort((x, y) => x - y);
  const pairs = [],
    excluded = [];
  for (const seed of seeds) {
    const left = maps[0].get(seed) ?? [],
      right = maps[1].get(seed) ?? [];
    let reason;
    if (!Number.isSafeInteger(seed)) reason = 'invalid seed';
    else if (left.length > 1 || right.length > 1) reason = 'duplicate seed';
    else if (left.length === 0 || right.length === 0) reason = 'unpaired seed';
    if (reason) {
      excluded.push({ seed, reason });
      continue;
    }
    const [x] = left,
      [y] = right;
    try {
      assert.deepEqual(
        [x.profile_0, x.profile_1, y.profile_0, y.profile_1],
        [baseline, strategist, strategist, baseline],
        'unexpected profile layout',
      );
      pairs.push({ seed, x, y, a: outcome(x), b: outcome(y) });
    } catch (error) {
      excluded.push({ seed, reason: error.message });
    }
  }
  return { pairs, excluded };
};
const summarize = (pairs) => {
  const matrix = labels.map(() => labels.map(() => 0));
  let g = 0,
    l = 0,
    improved = 0,
    worsened = 0;
  for (const { a, b } of pairs) {
    const i = labels.indexOf(a),
      j = labels.indexOf(b);
    matrix[i][j]++;
    g += Number(a !== 'S' && b === 'S');
    l += Number(a === 'S' && b !== 'S');
    improved += Number(j > i);
    worsened += Number(j < i);
  }
  const n = pairs.length;
  const delta = n ? (g - l) / n : null;
  const se = n > 1 ? Math.sqrt((g + l - n * delta ** 2) / (n * (n - 1))) : null;
  const changed = improved + worsened;
  const winsA = matrix[2].reduce((x, y) => x + y, 0);
  const winsB = matrix.reduce((sum, row) => sum + row[2], 0);
  assert.equal(
    matrix.flat().reduce((x, y) => x + y, 0),
    n,
  );
  if (n) assert.ok(Math.abs(delta - (winsB / n - winsA / n)) < 1e-12);
  let term = 2 ** -(g + l),
    tail = term;
  for (let k = 1; k <= Math.min(g, l); k++) {
    term *= (g + l - k + 1) / k;
    tail += term;
  }
  return {
    n,
    matrix,
    changed,
    changedFraction: n ? changed / n : null,
    improved,
    worsened,
    unchanged: n - changed,
    net: improved - worsened,
    winsA,
    winsB,
    g,
    l,
    delta,
    se,
    sigma: se > 0 ? delta / se : null,
    exactP: n ? Math.min(1, 2 * tail) : null,
  };
};

// Контроли различают победителя-профиль и номер стороны, сохраняют ничьи.
const fixture = (seed, side, result) => ({
  world_seed: seed,
  kind: 'arena',
  ticks: 36000,
  profile_0: side ? strategist : baseline,
  profile_1: side ? baseline : strategist,
  winner: result === 'D' ? null : result === 'S' ? (side ? 0 : 1) : side ? 1 : 0,
  end_reason: result === 'D' ? 'timeout' : 'base-destroyed',
});
for (const a of labels)
  for (const b of labels) {
    const p = pair([fixture(1, 0, a)], [fixture(1, 1, b)]);
    assert.deepEqual(
      p.pairs.map(({ a, b }) => [a, b]),
      [[a, b]],
    );
    const s = summarize(p.pairs);
    assert.equal(s.changed, Number(a !== b));
    assert.equal(s.g, Number(a !== 'S' && b === 'S'));
    assert.equal(s.l, Number(a === 'S' && b !== 'S'));
  }
const empty = pair([fixture(1, 0, 'S')], [fixture(2, 1, 'S')]);
assert.equal(empty.excluded.length, 2);
assert.equal(summarize(empty.pairs).delta, null);
assert.equal(summarize(empty.pairs).se, null);
assert.equal(pair([fixture(1, 0, 'S'), fixture(1, 0, 'B')], [fixture(1, 1, 'S')]).pairs.length, 0);
assert.equal(pair([fixture(null, 0, 'S')], [fixture(null, 1, 'S')]).pairs.length, 0);
assert.throws(() => outcome({ ...fixture(1, 0, 'D'), winner: 0 }));
assert.throws(() => outcome({ ...fixture(1, 0, 'S'), end_reason: null }));
assert.throws(() => outcome({ ...fixture(1, 0, 'S'), winner: 2 }));
const same = summarize(['S', 'B', 'D'].map((a) => ({ a, b: a })));
assert.equal(same.se, 0);
assert.equal(same.sigma, null);
assert.equal(same.exactP, 1);

const sources = [];
for (const run of ['33105088891', '33425774671']) {
  const path = `C:/src/dezintegra/TD_Game/.matchlog/pipeline-${run}/.matchlog/arena.sqlite`;
  const sha256 = await hash(path);
  const db = new DatabaseSync(path, { readOnly: true });
  const schema = db.prepare("select sql from sqlite_schema where name='match'").get().sql;
  const rows = db.prepare(sql).all();
  db.close();
  assert.equal(await hash(path), sha256, 'source changed during read');
  const groups = group(rows);
  sources.push({
    run,
    path,
    sha256,
    schema,
    rows,
    count: rows.length,
    uniqueValidSeeds: [...groups.keys()].filter(Number.isSafeInteger).length,
    duplicates: [...groups].filter(([, rows]) => rows.length > 1).map(([seed]) => seed),
    invalidSeeds: rows.filter((m) => !Number.isSafeInteger(m.world_seed)).map((m) => m.match_id),
    revisions: [...new Set(rows.map((m) => m.git_sha))],
    dirty: [...new Set(rows.map((m) => m.git_dirty))],
    outcomes: labels.map((o) => rows.filter((m) => outcome(m) === o).length),
    timeoutTicks: [...new Set(rows.filter((m) => m.end_reason === 'timeout').map((m) => m.ticks))],
  });
}
const { pairs, excluded } = pair(sources[0].rows, sources[1].rows);
const summary = summarize(pairs);
const only = (a, b) =>
  a.rows.filter((x) => !b.rows.some((y) => y.world_seed === x.world_seed)).map((x) => x.world_seed);
const metadata = {
  node: process.version,
  sources: sources.map(({ rows, ...rest }) => rest),
  onlyA: only(sources[0], sources[1]),
  onlyB: only(sources[1], sources[0]),
  excluded,
  sameSideAiSeeds: pairs.filter(
    ({ x, y }) => x.ai_seed_0 === y.ai_seed_0 && x.ai_seed_1 === y.ai_seed_1,
  ).length,
  sameProfileAiSeeds: pairs.filter(
    ({ x, y }) => x.ai_seed_0 === y.ai_seed_1 && x.ai_seed_1 === y.ai_seed_0,
  ).length,
  aiSeedFormulaMatches: sources.map(
    ({ rows }) =>
      rows.filter(
        (m) =>
          m.ai_seed_0 === (m.world_seed ^ 0x5bf03635) &&
          m.ai_seed_1 === (m.world_seed ^ 0x2f6e1a77),
      ).length,
  ),
  summary,
};
assert.deepEqual(
  sources.map((s) => s.outcomes),
  [
    [30, 7, 23],
    [30, 16, 14],
  ],
);
const header =
  '| seed | match_id A | match_id B | 0/1 A | 0/1 B | winner A | reason A | winner B | reason B | A | B | переход |';
const table = [
  header,
  '| --- | --- | --- | --- | --- | --- | --- | --- | --- | --- | --- | --- |',
  ...pairs.map(
    ({ seed, x, y, a, b }) =>
      `| ${seed} | ${x.match_id} | ${y.match_id} | ${x.profile_0}/${x.profile_1} | ${y.profile_0}/${y.profile_1} | ${x.winner ?? 'null'} | ${x.end_reason} | ${y.winner ?? 'null'} | ${y.end_reason} | ${a} | ${b} | ${a}→${b} |`,
  ),
].join('\n');
writeFileSync('.matchlog/paired-table.md', table + '\n');

const cells = Array(9).fill(0);
let gain = 0,
  loss = 0;
const savedRows = readFileSync('.matchlog/paired-table.md', 'utf8').trim().split('\n').slice(2);
for (const line of savedRows) {

}
assert.deepEqual(cells, summary.matrix.flat());
assert.deepEqual([gain, loss, savedRows.length], [summary.g, summary.l, summary.n]);
assert.deepEqual(

assert.deepEqual(

);
writeFileSync('.matchlog/paired-result.json', JSON.stringify(metadata, null, 2) + '\n');
console.log(JSON.stringify(metadata, null, 2));
console.log('Контроли и независимый пересчёт таблицы: OK');
